"""Frozen DINOv3 last-four-CLS + final patch-mean CIFAR-100 linear probe."""
import os
os.environ.setdefault("CUBLAS_WORKSPACE_CONFIG", ":4096:8")
import argparse
import hashlib
import json
import random
import time
from pathlib import Path


def write_json(path, value):
    path = Path(path)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_text(json.dumps(value, indent=2, ensure_ascii=False), encoding="utf-8")
    tmp.replace(path)


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8 * 1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def seed_all(seed):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)


def split_indices(labels, seed):
    rng = np.random.default_rng(seed)
    train, val = [], []
    for label in range(100):
        indices = np.flatnonzero(labels == label)
        if len(indices) != 500:
            raise ValueError("Expected the complete CIFAR-100 training split")
        rng.shuffle(indices)
        val.extend(indices[:50])
        train.extend(indices[50:])
    return np.array(train), np.array(val)


def load_backbone(args):
    from transformers import AutoModel
    model, info = AutoModel.from_pretrained(
        args.model_dir, local_files_only=True, attn_implementation="sdpa",
        output_loading_info=True,
    )
    if info.get("missing_keys") or info.get("unexpected_keys") or info.get("mismatched_keys") or info.get("error_msgs"):
        raise RuntimeError(f"Checkpoint loading mismatch: {info}")
    model.requires_grad_(False).eval().cuda()
    return model


def processor_transform(args):
    from torchvision import transforms as T
    from torchvision.transforms import InterpolationMode
    cfg = json.loads((Path(args.model_dir) / "preprocessor_config.json").read_text())
    if cfg.get("resample") != 2 or cfg.get("size") != {"height": 224, "width": 224}:
        raise ValueError("This recipe requires the inspected 224x224 bilinear processor")
    if cfg.get("rescale_factor") != 1 / 255 or not cfg.get("do_normalize") or cfg.get("do_center_crop"):
        raise ValueError("Unexpected preprocessing; review the recipe")
    # Explicit PIL bilinear preprocessing; no stochastic augmentation.
    return T.Compose([
        T.Resize((224, 224), interpolation=InterpolationMode.BILINEAR),
        T.ToTensor(), T.Normalize(cfg["image_mean"], cfg["image_std"]),
    ])


def extract_batch(model, pixels):
    with torch.inference_mode(), torch.autocast("cuda", dtype=torch.float16):
        # The official intermediate-layer path applies the shared final norm to
        # each selected block output before taking CLS. Capture only these tiny
        # normalized CLS tensors instead of retaining every full hidden state.
        cls_tokens = []

        def capture_cls(module, inputs, output):
            cls_tokens.append(model.norm(output[:, :1])[:, 0].float())

        if len(model.layer) < 4:
            raise ValueError("At least four transformer blocks are required")
        handles = [layer.register_forward_hook(capture_cls) for layer in model.layer[-4:]]
        try:
            hidden = model(pixel_values=pixels.cuda(non_blocking=True)).last_hidden_state
        finally:
            for handle in handles:
                handle.remove()
        if len(cls_tokens) != 4:
            raise RuntimeError("Did not capture precisely the last four CLS tokens")
        patches = hidden[:, 1 + model.config.num_register_tokens:]
        if patches.shape[1] != 196:
            raise RuntimeError("Expected 196 image patches; register tokens must be excluded")
        features = torch.cat([*cls_tokens, patches.float().mean(dim=1)], dim=1)
    if not torch.isfinite(features).all():
        raise RuntimeError("Non-finite backbone features; review FP16 before proceeding")
    return features.cpu()


def extract(args, model, train):
    from torchvision.datasets import CIFAR100
    from torch.utils.data import DataLoader, Subset
    split = "train" if train else "test"
    dataset = CIFAR100(args.data_dir, train=train, download=True, transform=processor_transform(args))
    folder = Path(args.run_dir) / "features" / split
    folder.mkdir(parents=True, exist_ok=True)
    result, labels = [], []
    started = time.time()
    for begin in range(0, len(dataset), 2048):
        stop = min(begin + 2048, len(dataset))
        dest = folder / f"{begin:05d}-{stop:05d}.pt"
        if dest.exists():
            saved = torch.load(dest, weights_only=True, map_location="cpu")
        else:
            loader = DataLoader(Subset(dataset, range(begin, stop)), batch_size=args.batch_size,
                                shuffle=False, num_workers=args.workers, pin_memory=True)
            features, targets = [], []
            for pixels, target in loader:
                features.append(extract_batch(model, pixels).half())
                targets.append(target)
            saved = {"features": torch.cat(features), "labels": torch.cat(targets)}
            tmp = dest.with_suffix(".tmp")
            torch.save(saved, tmp)
            tmp.replace(dest)
        if saved["features"].shape != (stop - begin, model.config.hidden_size * 5):
            raise RuntimeError(f"Invalid cached feature shape: {dest}")
        if not torch.equal(saved["labels"], torch.tensor(dataset.targets[begin:stop])):
            raise RuntimeError(f"Cached labels do not match the dataset: {dest}")
        result.append(saved["features"])
        labels.append(saved["labels"])
        print(json.dumps({"stage": "extract", "split": split, "done": stop,
                          "total": len(dataset), "elapsed_seconds": round(time.time() - started, 1)}), flush=True)
    return torch.cat(result).float().cuda(), torch.cat(labels).cuda()


def accuracy(head, x, y):
    correct, loss = 0, 0.0
    with torch.no_grad():
        for begin in range(0, len(x), 2048):
            logits = head(x[begin:begin + 2048])
            target = y[begin:begin + 2048]
            correct += (logits.argmax(1) == target).sum().item()
            loss += torch.nn.functional.cross_entropy(logits, target, reduction="sum").item()
    return 100 * correct / len(y), loss / len(y)


def fit_head(x, y, args, lr, wd, epochs, validation=None):
    seed_all(args.seed)
    head = torch.nn.Linear(x.shape[1], 100).cuda()
    optimizer = torch.optim.SGD(head.parameters(), lr=lr, momentum=0.9, weight_decay=wd)
    history, best = [], None
    for epoch in range(1, epochs + 1):
        head.train()
        order = torch.randperm(len(x), device="cuda")
        for batch in order.split(1024):
            optimizer.zero_grad(set_to_none=True)
            loss = torch.nn.functional.cross_entropy(head(x[batch]), y[batch])
            if not torch.isfinite(loss):
                raise RuntimeError("Non-finite classifier loss")
            loss.backward()
            optimizer.step()
        head.eval()
        if validation is not None:
            top1, val_loss = accuracy(head, *validation)
            row = {"lr": lr, "weight_decay": wd, "epoch": epoch,
                   "val_top1_percent": top1, "val_loss": val_loss}
            history.append(row)
            if best is None or (top1, -val_loss) > (best["val_top1_percent"], -best["val_loss"]):
                best = row
        if epoch == 1 or epoch % 10 == 0 or epoch == epochs:
            print(json.dumps({"stage": "head_fit", "lr": lr, "weight_decay": wd,
                              "epoch": epoch, "epochs": epochs,
                              **(history[-1] if validation is not None else {})}), flush=True)
    return head, best, history


def smoke(args):
    from torchvision.datasets import CIFAR100
    from torch.utils.data import DataLoader, Subset
    model = load_backbone(args)
    transform = processor_transform(args)
    dataset = CIFAR100(args.data_dir, train=True, download=True, transform=transform)
    pixels, targets = next(iter(DataLoader(Subset(dataset, range(args.batch_size)),
                                         batch_size=args.batch_size, num_workers=0)))
    # Check the explicit preprocessing against the model's own image processor.
    from transformers import AutoImageProcessor
    from PIL import Image
    processor = AutoImageProcessor.from_pretrained(args.model_dir, local_files_only=True)
    original = Image.fromarray(dataset.data[0])
    reference = processor(images=original, return_tensors="pt")["pixel_values"]
    max_diff = (transform(original).unsqueeze(0) - reference).abs().max().item()
    if max_diff > 0.06:
        raise RuntimeError(f"Unexpected processor mismatch: max abs difference {max_diff}")
    # PIL and tensor resize kernels may differ slightly; record the actual difference.
    extract_batch(model, pixels)  # Warm up kernels before timing.
    torch.cuda.reset_peak_memory_stats()
    torch.cuda.synchronize()
    started = time.perf_counter()
    features = extract_batch(model, pixels)
    torch.cuda.synchronize()
    elapsed = time.perf_counter() - started
    head = torch.nn.Linear(features.shape[1], 100).cuda()
    loss = torch.nn.functional.cross_entropy(head(features.cuda()), targets.cuda())
    loss.backward()
    if not torch.isfinite(loss):
        raise RuntimeError("Non-finite smoke-test loss")
    result = {"smoke_passed": True, "batch_size": args.batch_size,
              "feature_shape": list(features.shape), "loss": loss.item(),
              "processor_max_abs_difference": max_diff,
              "forward_seconds": elapsed, "images_per_second": len(pixels) / elapsed,
              "peak_allocated_gib": torch.cuda.max_memory_allocated() / 1024**3,
              "gpu": torch.cuda.get_device_name(0)}
    write_json(Path(args.run_dir) / "smoke.json", result)
    print(json.dumps(result), flush=True)


def run(args):
    root = Path(args.run_dir)
    identity = {
        "recipe": "dinov3-cifar100-frozen-last4cls-patchmean-pil224-v1",
        "feature_dimension": 5120,
        "cls_block_indices_zero_based": [20, 21, 22, 23],
        "cls_normalization": "shared_final_layernorm_per_block",
        "weights_sha256": sha256(Path(args.model_dir) / "model.safetensors"),
        "config_sha256": sha256(Path(args.model_dir) / "config.json"),
        "processor_sha256": sha256(Path(args.model_dir) / "preprocessor_config.json"),
        "script_sha256": sha256(Path(__file__)),
        "seed": args.seed, "epochs": args.epochs,
        "learning_rates": args.learning_rates, "weight_decays": args.weight_decays,
        "feature_cache_dtype": "float16", "classifier_dtype": "float32",
    }
    identity_file = root / "identity.json"
    if identity_file.exists() and json.loads(identity_file.read_text()) != identity:
        raise RuntimeError("Run identity changed. Choose a new --run-dir to avoid mixing experiments")
    write_json(identity_file, identity)
    if (root / "result.json").exists():
        print((root / "result.json").read_text(), flush=True)
        return
    import platform
    import subprocess
    import transformers
    import torchvision
    write_json(root / "environment.json", {
        "python": platform.python_version(), "torch": torch.__version__,
        "torchvision": torchvision.__version__, "transformers": transformers.__version__,
        "numpy": np.__version__, "cuda_runtime": torch.version.cuda,
        "gpu": torch.cuda.get_device_name(0), "arguments": vars(args),
    })
    with open(root / "packages.txt", "w", encoding="utf-8") as handle:
        subprocess.run([sys.executable, "-m", "pip", "freeze"], stdout=handle, check=True)
    model = load_backbone(args)
    x, y = extract(args, model, train=True)
    train_ids, val_ids = split_indices(y.cpu().numpy(), args.seed)
    write_json(root / "split.json", {"train_indices": train_ids.tolist(), "val_indices": val_ids.tolist()})
    xt, yt = x[train_ids], y[train_ids]
    xv, yv = x[val_ids], y[val_ids]
    mean, std = xt.mean(0), xt.std(0, unbiased=False).clamp_min(1e-6)
    xt, xv = (xt - mean) / std, (xv - mean) / std
    best, history = None, []
    for lr in args.learning_rates:
        for wd in args.weight_decays:
            _, candidate, rows = fit_head(xt, yt, args, lr, wd, args.epochs, (xv, yv))
            history.extend(rows)
            if best is None or (candidate["val_top1_percent"], -candidate["val_loss"]) > (best["val_top1_percent"], -best["val_loss"]):
                best = candidate
            write_json(root / "validation_history.json", history)
            write_json(root / "selection.json", best)
    del xt, xv, yt, yv
    # Refit on every official training example with hyperparameters fixed by validation.
    mean, std = x.mean(0), x.std(0, unbiased=False).clamp_min(1e-6)
    head, _, _ = fit_head((x - mean) / std, y, args, best["lr"], best["weight_decay"], best["epoch"])
    torch.save({"head": {k: v.cpu() for k, v in head.state_dict().items()},
                "feature_mean": mean.cpu(), "feature_std": std.cpu(),
                "selection": best, "identity": identity}, root / "classifier.pt")
    # No official test features or test labels are used for model selection.
    test_x, test_y = extract(args, model, train=False)
    top1, test_loss = accuracy(head, (test_x - mean) / std, test_y)
    result = {"test_top1_percent": top1, "test_loss": test_loss,
              "test_examples": len(test_y), "correct": round(top1 * len(test_y) / 100),
              "refit_train_examples": len(y), "selection": best,
              "protocol": identity["recipe"], "seed": args.seed}
    write_json(root / "result.json", result)
    print(json.dumps(result), flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--mode", choices=["smoke", "run"], default="smoke")
    parser.add_argument("--model-dir", required=True)
    parser.add_argument("--data-dir", default="data")
    parser.add_argument("--run-dir", default="runs/vitl16_last4_linear_seed42")
    parser.add_argument("--batch-size", type=int, default=16)
    parser.add_argument("--workers", type=int, default=4)
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--epochs", type=int, default=100)
    parser.add_argument("--learning-rates", type=float, nargs="+", default=[0.01, 0.03, 0.1])
    parser.add_argument("--weight-decays", type=float, nargs="+", default=[0.0, 0.001])
    args = parser.parse_args()
    if args.batch_size < 1 or args.epochs < 1 or args.workers < 0:
        parser.error("Invalid batch size, epochs, or worker count")
    Path(args.run_dir).mkdir(parents=True, exist_ok=True)
    # Linux advisory lock is released automatically when this process exits.
    import fcntl
    run_lock = open(Path(args.run_dir) / "process.lock", "a")
    try:
        fcntl.flock(run_lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        raise RuntimeError("Another experiment process is already using this run directory")
    global torch, np, sys
    import sys
    import torch
    import numpy as np
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA GPU required; check the driver and PyTorch installation")
    torch.set_num_threads(4)
    torch.backends.cudnn.benchmark = False
    torch.backends.cudnn.deterministic = True
    torch.use_deterministic_algorithms(True)
    seed_all(args.seed)
    (smoke if args.mode == "smoke" else run)(args)


if __name__ == "__main__":
    main()
