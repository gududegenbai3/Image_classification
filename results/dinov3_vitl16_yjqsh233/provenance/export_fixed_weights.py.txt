def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + "\n", encoding="utf-8")


def sha(path):
    return original.sha256(path)

# Compare all repository PNG blob hashes without downloading or changing images.
tree = json.loads(Path(args.tree).read_text())
repo_images = {item["path"]: item["sha"] for item in tree["tree"]
               if item["type"] == "blob" and item["path"].endswith(".png")
               and item["path"].startswith(("dataset/train/", "dataset/test/"))}
matches, checked, mismatches = 0, 0, []
for split_name, dataset in [("train", training), ("test", testing)]:
    with (data_dir / "cifar-100-python" / split_name).open("rb") as handle:
        names = pickle.load(handle, encoding="bytes")[b"filenames"]
    for index, filename in enumerate(names):
        filename = filename.decode() if isinstance(filename, bytes) else filename
        relative = "dataset/" + split_name + "/" + dataset.classes[dataset.targets[index]] + "/" + filename
        buffer = io.BytesIO()
        Image.fromarray(dataset.data[index]).save(buffer, format="PNG")
        png = buffer.getvalue()
        digest = hashlib.sha1(b"blob " + str(len(png)).encode() + b"\0" + png).hexdigest()
        matches += int(repo_images.get(relative) == digest)
        checked += 1
        if repo_images.get(relative) != digest and len(mismatches) < 10:
            mismatches.append({"path": relative, "generated_git_blob_sha": digest,
                               "repository_git_blob_sha": repo_images.get(relative)})
        if checked % 10000 == 0:
            print(json.dumps({"stage": "dataset_identity", "checked": checked, "matched": matches}), flush=True)
assert checked == len(repo_images) == 60000
dataset_identity = {"checked_png_images": checked, "exact_git_blob_matches": matches,
                    "all_images_byte_identical": matches == checked, "first_mismatches": mismatches,
                    "method": "Canonical CIFAR pixels and original filenames encoded with PIL PNG, compared to upstream Git blob SHA1"}
write(stage / "results" / key / "dataset_identity.json", dataset_identity)


def compact_state(old):
    if "head" in old:
        return {"head.weight": old["head"]["weight"], "head.bias": old["head"]["bias"],
                "feature_mean": old["feature_mean"], "feature_std": old["feature_std"]}
    return old["model"]


def score(model, dataset, indices=None, cached=None, batch=64):
    totals = metrics_api.ClassificationMetrics(100)
    labels, predictions = [], []
    model.eval()
    with torch.inference_mode():
        if cached is not None:
            parts = [torch.load(path, weights_only=True, map_location="cpu")
                     for path in sorted(cached.glob("*.pt"))]
            x = torch.cat([part["features"] for part in parts]).float().cuda()
            y = torch.cat([part["labels"] for part in parts]).cuda()
            x = (x - model.feature_mean) / model.feature_std
            iterator = ((x[b:b + 2048], y[b:b + 2048]) for b in range(0, len(x), 2048))
        else:
            subset = Subset(dataset, indices) if indices is not None else dataset
            iterator = DataLoader(subset, batch_size=batch, shuffle=False, num_workers=4, pin_memory=True)
        for inputs, targets in iterator:
            inputs, targets = inputs.cuda(non_blocking=True), targets.cuda(non_blocking=True)
            logits = model.head(inputs) if cached is not None else adapter.predict(model, inputs)
            loss = torch.nn.functional.cross_entropy(logits.float(), targets, reduction="sum")
            predicted = logits.argmax(1)
            truth, guessed = targets.tolist(), predicted.tolist()
            totals.update(truth, guessed, loss.item())
            labels.extend(truth)
            predictions.extend(guessed)
    return totals.compute(), {"targets": labels, "predictions": predictions}

                               warm_start_head_epochs=23,
